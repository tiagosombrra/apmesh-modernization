cmake_minimum_required(VERSION 3.21)
if(CMAKE_SOURCE_DIR STREQUAL CMAKE_CURRENT_SOURCE_DIR)
  project(apmesh_model_ownership LANGUAGES CXX)
endif()
get_filename_component(default_root "${CMAKE_CURRENT_LIST_DIR}/../.." ABSOLUTE)
set(AP_MESH_MODEL_SOURCE_ROOT "${default_root}" CACHE PATH "Actual Model source snapshot")
option(AP_MESH_MODEL_ASAN "Instrument the actual Model ownership probe" OFF)
option(AP_MESH_MODEL_ORIGINAL "Build original raw adoption observations only" OFF)
set(model_sources data/model.cpp data/mesh/mesh_adaptive.cpp data/mesh/sub_mesh.cpp
  data/node_adaptive.cpp data/element_adaptive.cpp data/triangle_adaptive.cpp
  data/point_adaptive.cpp data/vector_adaptive.cpp data/patch/patch.cpp
  data/geometry.cpp data/curve/curve_adaptive.cpp data/curve/curve_adaptive_parametric.cpp
  data/curve/curve_adaptive_parametric_bezier.cpp crab_mesh/numerical/function.cpp
  crab_mesh/numerical/bisection_equation_root.cpp)
list(TRANSFORM model_sources PREPEND "${AP_MESH_MODEL_SOURCE_ROOT}/src/")
add_executable(model_ownership_probe probe.cpp ${model_sources})
target_include_directories(model_ownership_probe PRIVATE "${AP_MESH_MODEL_SOURCE_ROOT}/include")
target_compile_features(model_ownership_probe PRIVATE cxx_std_17)
target_compile_definitions(model_ownership_probe PRIVATE _USE_MATH_DEFINES)
if(AP_MESH_MODEL_ORIGINAL)
  target_compile_definitions(model_ownership_probe PRIVATE AP_MESH_MODEL_ORIGINAL=1)
endif()
if(MSVC)
  target_compile_options(model_ownership_probe PRIVATE /utf-8 /W4 /EHsc /permissive-)
  target_compile_definitions(model_ownership_probe PRIVATE _ITERATOR_DEBUG_LEVEL=0)
  if(AP_MESH_MODEL_ASAN)
    set_property(TARGET model_ownership_probe PROPERTY MSVC_RUNTIME_LIBRARY MultiThreadedDLL)
    target_compile_options(model_ownership_probe PRIVATE /fsanitize=address /Zi)
    target_link_options(model_ownership_probe PRIVATE /INCREMENTAL:NO)
  else()
    set_property(TARGET model_ownership_probe PROPERTY MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
  endif()
elseif(MINGW AND AP_MESH_STATIC_MINGW_RUNTIME)
  target_link_options(model_ownership_probe PRIVATE -static -static-libgcc -static-libstdc++)
endif()
