find_package(Python3 3.10 REQUIRED COMPONENTS Interpreter)
get_filename_component(default_edge_root "${CMAKE_CURRENT_LIST_DIR}/../.." ABSOLUTE)
set(AP_MESH_FIRST_EDGE_ROOT "${default_edge_root}" CACHE PATH "Native source tree")
set(AP_MESH_FIRST_EDGE_ORIGINAL_SOURCE "" CACHE FILEPATH "Optional pinned original front source")
set(AP_MESH_FIRST_EDGE_STALE_SOURCE "" CACHE FILEPATH "Optional deliberately stale pending adjacency")
set(AP_MESH_FIRST_EDGE_BROKEN_SOURCE "" CACHE FILEPATH "Optional deliberately invalid pending deleter")
option(AP_MESH_FIRST_EDGE_ASAN "Instrument the bounded first-edge adoption probe" OFF)
set(edge_sources crab_mesh/aft/shape.cpp crab_mesh/aft/vertex.cpp crab_mesh/aft/edge.cpp
  crab_mesh/aft/face.cpp crab_mesh/aft/boundary.cpp crab_mesh/aft/quadtree_cell.cpp crab_mesh/aft/quadtree.cpp

list(TRANSFORM edge_sources PREPEND "${AP_MESH_FIRST_EDGE_ROOT}/src/")
add_library(first_edge_native STATIC ${edge_sources})
function(configure_edge_target target)
  set_target_properties(${target} PROPERTIES CXX_STANDARD 17 CXX_STANDARD_REQUIRED YES CXX_EXTENSIONS OFF)
  target_include_directories(${target} PRIVATE "${AP_MESH_FIRST_EDGE_ROOT}/include"
    "${AP_MESH_FIRST_EDGE_ROOT}/src/crab_mesh/aft" "${CMAKE_CURRENT_SOURCE_DIR}")
  if(MSVC)
    target_compile_options(${target} PRIVATE /utf-8 /W4 /EHsc)
    target_compile_definitions(${target} PRIVATE $<$<CONFIG:Debug>:_ITERATOR_DEBUG_LEVEL=0>)
    if(AP_MESH_FIRST_EDGE_ASAN)
      target_compile_options(${target} PRIVATE /fsanitize=address /Zi)
      target_link_options(${target} PRIVATE /INCREMENTAL:NO)
    endif()
  elseif(MINGW AND AP_MESH_STATIC_MINGW_RUNTIME)
    target_link_options(${target} PRIVATE -static -static-libgcc -static-libstdc++)
  endif()
endfunction()
configure_edge_target(first_edge_native)
function(add_edge_probe target source)
  set(backend "${CMAKE_CURRENT_BINARY_DIR}/${target}_backend.cpp")
  add_custom_command(OUTPUT "${backend}"
    COMMAND "${Python3_EXECUTABLE}" "${AP_MESH_FIRST_EDGE_ROOT}/tools/prepare_first_edge_probe.py"
      --root "${AP_MESH_FIRST_EDGE_ROOT}" --source "${source}" --output "${backend}"
      --metadata "${CMAKE_CURRENT_BINARY_DIR}/${target}_observer.json"
    DEPENDS "${source}" "${AP_MESH_FIRST_EDGE_ROOT}/tools/prepare_first_edge_probe.py" VERBATIM)
  add_executable(${target} probe.cpp uaf_control.cpp leak_control.cpp "${backend}")
  configure_edge_target(${target})
  target_link_libraries(${target} PRIVATE first_edge_native)
endfunction()
add_edge_probe(first_edge_probe "${AP_MESH_FIRST_EDGE_ROOT}/src/crab_mesh/aft/advancing_front.cpp")
