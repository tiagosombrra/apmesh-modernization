function(configure_marker_target target)
  set_target_properties(${target} PROPERTIES CXX_STANDARD 17 CXX_STANDARD_REQUIRED YES CXX_EXTENSIONS OFF)
  target_include_directories(${target} PRIVATE "${AP_MESH_MARKER_ROOT}/include"
    "${AP_MESH_MARKER_ROOT}/src/crab_mesh/aft" "${CMAKE_CURRENT_SOURCE_DIR}")
  if(MSVC)
    target_compile_options(${target} PRIVATE /utf-8 /W4 /EHsc)
    target_compile_definitions(${target} PRIVATE $<$<CONFIG:Debug>:_ITERATOR_DEBUG_LEVEL=0>)
    if(AP_MESH_MARKER_ASAN)
      target_compile_options(${target} PRIVATE /fsanitize=address /Zi)
      target_link_options(${target} PRIVATE /INCREMENTAL:NO)
    endif()
  elseif(MINGW AND AP_MESH_STATIC_MINGW_RUNTIME)
    target_link_options(${target} PRIVATE -static -static-libgcc -static-libstdc++)
  endif()
endfunction()
configure_marker_target(front_marker_native)
function(add_marker_probe target source)
  set(backend "${CMAKE_CURRENT_BINARY_DIR}/${target}_backend.cpp")
  add_custom_command(OUTPUT "${backend}"
    COMMAND "${Python3_EXECUTABLE}" "${AP_MESH_MARKER_ROOT}/tools/prepare_marker_probe.py"
      --root "${AP_MESH_MARKER_ROOT}" --source "${source}" --output "${backend}"
      --metadata "${CMAKE_CURRENT_BINARY_DIR}/${target}_observer.json"
    DEPENDS "${source}" "${AP_MESH_MARKER_ROOT}/tools/prepare_marker_probe.py" VERBATIM)
  add_executable(${target} probe.cpp uaf_control.cpp "${backend}")
  configure_marker_target(${target})
  target_link_libraries(${target} PRIVATE front_marker_native)
endfunction()
add_marker_probe(marker_probe "${AP_MESH_MARKER_ROOT}/src/crab_mesh/aft/advancing_front.cpp")
